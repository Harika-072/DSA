import java.util.*;
import java.lang.*;
import java.io.*;

class Codechef
{
	public static void main (String[] args) throws java.lang.Exception
	{
		// your code goes here
		Scanner sc=new Scanner(System.in);
		int n=sc.nextInt();
		LinkedHashSet<String> ans=new LinkedHashSet<>();
		for(int i=0;i<n;i++){
		    ans.add(sc.next());
		}
		for(String i:ans){
		    System.out.println(i);
		}

	}
}
