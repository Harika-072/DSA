import java.util.ArrayList;
import java.util.Collections;
import java.util.Scanner;

class Codechef {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        // Step 1: Take input for number of languages
        int n=sc.nextInt();
            sc.nextLine();

        // Step 2: Create ArrayList to store language names
            ArrayList<String> language=new ArrayList<>();

        // Step 3: Read n languages from user
        for(int i=0;i<n;i++){
            language.add(sc.nextLine());
        }



        // Step 4: Sort the list
            Collections.sort(language);

        // Step 5: Print first and last elements
        System.out.println("First Language: "+language.get(0));
        System.out.println("Last Language: "+ language.get(language.size()-1));

        
        sc.close();
    }
}
